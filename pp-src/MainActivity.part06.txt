    private class ConfettiView extends View {
        Paint p=new Paint(Paint.ANTI_ALIAS_FLAG);List<Particle> parts=new ArrayList<>();ValueAnimator anim;Random r=new Random();
        ConfettiView(Context c){super(c);setClickable(false);for(int i=0;i<42;i++)parts.add(new Particle());}
        void start(){anim=ValueAnimator.ofFloat(0,1);anim.setDuration(1800);anim.addUpdateListener(a->{float f=(float)a.getAnimatedValue();for(Particle q:parts){q.y=q.sy+f*dp(420)+f*f*dp(180);q.x=q.sx+(float)Math.sin(f*6+q.phase)*dp(38);}setAlpha(1-f*.75f);invalidate();});anim.start();}
        @Override protected void onDraw(Canvas c){super.onDraw(c);int[] cols={PRIMARY,GREEN,Color.rgb(244,114,182),Color.rgb(251,191,36)};for(int i=0;i<parts.size();i++){Particle q=parts.get(i);p.setColor(cols[i%cols.length]);c.save();c.rotate(q.rot,q.x,q.y);c.drawRoundRect(new RectF(q.x,q.y,q.x+dp(7),q.y+dp(14)),dp(2),dp(2),p);c.restore();}}
        class Particle{float sx=r.nextInt(Math.max(1,getResources().getDisplayMetrics().widthPixels));float sy=-r.nextInt(dp(180));float x=sx,y=sy,phase=r.nextFloat()*6,rot=r.nextFloat()*180;}
    }
